> **For agentic workers:** Use superpowers-open:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为工业手册问答工作台增加语音输入（语音转文字填入输入框）与朗读回答（文字转语音播放）两项能力。

**Architecture:** 后端新增两个 FastAPI 端点代理 DashScope 语音能力（`qwen3-asr-flash` / `qwen3-tts-flash`），复用现有匿名限流与 `AppError` 约定；前端新增两个 composable 管理录音与播放状态机。文本清洗与文件头探测抽成纯函数以便单测。

**Tech Stack:** Python 3.11 / FastAPI / httpx / pytest；Vue 3 / TypeScript / vitest / MediaRecorder API

**设计依据：** `docs/specs/2026-11-12-voice-io-design.md`

## 关键前置事实（已实测，不要重新假设）

| 项 | 值 |
| --- | --- |
| ASR 模型 | `qwen3-asr-flash` |
| TTS 模型 | `qwen3-tts-flash` |
| 端点 | `https://dashscope.aliyuncs.com/api/v1/services/aigc/multimodal-generation/generation` |
| ASR 请求体 | `{"model":"qwen3-asr-flash","input":{"messages":[{"role":"user","content":[{"type":"audio","audio":"data:<mime>;base64,<b64>"}]}]}}` |
| ASR 响应路径 | `output.choices[0].message.content[0].text` |
| TTS 请求体 | `{"model":"qwen3-tts-flash","input":{"text":"<text>","voice":"Cherry","language_type":"Chinese"}}` |
| TTS 响应路径 | `output.audio.url`（24 小时有效的 OSS 链接） |
| 配置环境变量前缀 | **`IRA_`**（例如 `IRA_VOICE_ENABLED`），不是 `VOICE_` |
| 回答长度 | 历史中位 82、最大 424 字符，均低于 TTS 600 上限 |

**实测缺陷（必须防）：** ASR 传入 4004 字节随机数据时**不报错**，返回 `{"text":"啊。"}`。因此需要三道防线（文件头校验 + 最短时长 + 低置信度判定）。

```python
"""Deterministic text and audio helpers for the voice endpoints.

Everything here is a pure function so it can be tested without network or audio
hardware. The cleaning rules matter because raw answers carry citation markers
(``[1]``) that a TTS engine would read out loud as "square bracket one".
"""

from __future__ import annotations

import re

# Citation markers produced by the structured citation path.
_CITATION = re.compile(r"\[\d+\]")
# Markdown emphasis and code fences never belong in spoken text.
_MARKDOWN = re.compile(r"[*_`#>]+")
# Spoken forms for the units that actually appear in these manuals.
_UNIT_SPEECH = (
    ("°C", "摄氏度"),
    ("℃", "摄氏度"),
    ("°F", "华氏度"),
    ("mm/s", "毫米每秒"),
    ("m³/h", "立方米每小时"),
    ("mm", "毫米"),
    ("MPa", "兆帕"),
    ("kPa", "千帕"),
    ("rpm", "转每分钟"),
)
_WHITESPACE = re.compile(r"\s+")

_MIME_BY_KIND = {
    "wav": "audio/wav",
    "webm": "audio/webm",
    "ogg": "audio/ogg",
    "mp4": "audio/mp4",
}


def clean_for_speech(text: str, *, max_chars: int = 600) -> str:
    """Turn an API answer into text suitable for text-to-speech.

    Removes citation markers and markdown, expands units into their spoken
    Chinese form, collapses whitespace, and truncates to ``max_chars`` because
    the TTS model rejects longer input.
    """

    if not text:
        return ""
    cleaned = _CITATION.sub("", text)
    cleaned = _MARKDOWN.sub("", cleaned)
    for symbol, spoken in _UNIT_SPEECH:
        cleaned = cleaned.replace(symbol, spoken)
    cleaned = _WHITESPACE.sub(" ", cleaned).strip()
    return cleaned[:max_chars]


def is_low_confidence(text: str) -> bool:
    """Report whether an ASR result is too short to be a real question.

    The ASR model does not reject unusable audio: sending random bytes returns a
    short hallucinated phrase such as "啊。". Anything with two or fewer
    non-whitespace characters is treated as noise so the UI can ask the user to
    retry instead of pasting it into the input box.
    """

    return len(text.strip()) <= 2


def sniff_audio_kind(data: bytes) -> str | None:
    """Identify the container from its magic bytes.

    The upstream service ignores the declared MIME type, so the server must
    decide for itself whether the payload is plausibly audio before spending a
    billable ASR call on it.
    """

    if len(data) < 4:
        return None
    if data.startswith(b"RIFF"):
        return "wav"
    if data.startswith(b"\x1a\x45\xdf\xa3"):
        return "webm"
    if data.startswith(b"OggS"):
        return "ogg"
    if len(data) >= 12 and data[4:8] == b"ftyp":
        return "mp4"
    return None


def mime_for_kind(kind: str) -> str:
    """Map a sniffed container kind to the MIME type sent upstream."""

    return _MIME_BY_KIND.get(kind, "audio/wav")
```

```python
"""DashScope adapter for speech recognition and synthesis.

The router depends on the ``VoiceProvider`` protocol, not on this class, so tests
can substitute a fake and never touch the network or the paid quota.
"""

from __future__ import annotations

import base64
import logging
from dataclasses import dataclass, field
from typing import Any, Protocol

import httpx

logger = logging.getLogger("industrial_rag.voice_provider")

_MULTIMODAL_PATH = "/services/aigc/multimodal-generation/generation"


class VoiceProviderError(RuntimeError):
    """Raised for any upstream failure the router should map to 503."""


@dataclass(frozen=True, slots=True)
class TranscriptionResult:
    text: str
    language: str | None = None
    emotion: str | None = None
    raw: dict[str, Any] = field(default_factory=dict)


class VoiceProvider(Protocol):
    """Minimal surface the router needs."""

    async def transcribe(self, audio: bytes, mime: str) -> TranscriptionResult: ...

    async def synthesize(self, text: str) -> bytes: ...


class DashScopeVoiceProvider:
    """Calls the DashScope multimodal endpoint for ASR and TTS."""

    def __init__(
        self,
        *,
        api_key: str,
        base_url: str,
        model: str,
        tts_model: str,
        voice: str,
        client: httpx.AsyncClient | None = None,
        timeout: float = 60.0,
    ) -> None:
        self._api_key = api_key
        self._base_url = base_url.rstrip("/")
        self._model = model
        self._tts_model = tts_model
        self._voice = voice
        self._owns_client = client is None
        self._client = client or httpx.AsyncClient(timeout=timeout)

    async def aclose(self) -> None:
        if self._owns_client:
            await self._client.aclose()

    def _headers(self, *, sse: bool = False) -> dict[str, str]:
        headers = {
            "Authorization": f"Bearer {self._api_key}",
            "Content-Type": "application/json",
        }
        if sse:
            headers["X-DashScope-SSE"] = "enable"
        return headers

    async def _post(self, payload: dict[str, Any]) -> dict[str, Any]:
        url = self._base_url + _MULTIMODAL_PATH
        try:
            response = await self._client.post(url, headers=self._headers(), json=payload)
        except httpx.TimeoutException as error:
            raise VoiceProviderError(f"voice upstream timeout: {error}") from error
        except httpx.HTTPError as error:
            raise VoiceProviderError(f"voice upstream transport error: {error}") from error
        if response.status_code >= 400:
            raise VoiceProviderError(
                f"voice upstream HTTP {response.status_code}: {response.text[:200]}"
            )
        try:
            return response.json()
        except ValueError as error:
            raise VoiceProviderError("voice upstream returned non-JSON") from error

    async def transcribe(self, audio: bytes, mime: str) -> TranscriptionResult:
        encoded = base64.b64encode(audio).decode("ascii")
        payload = {
            "model": self._model,
            "input": {
                "messages": [
                    {
                        "role": "user",
                        "content": [
                            {"type": "audio", "audio": f"data:{mime};base64,{encoded}"}
                        ],
                    }
                ]
            },
        }
        body = await self._post(payload)
        try:
            message = body["output"]["choices"][0]["message"]
            content = message["content"]
            text = content[0]["text"]
        except (KeyError, IndexError, TypeError) as error:
            raise VoiceProviderError(
                "voice upstream response missing content text"
            ) from error
        annotations = message.get("annotations") or []
        language = annotations[0].get("language") if annotations else None
        emotion = annotations[0].get("emotion") if annotations else None
        return TranscriptionResult(
            text=str(text), language=language, emotion=emotion, raw=body
        )

    async def synthesize(self, text: str) -> bytes:
        payload = {
            "model": self._tts_model,
            "input": {"text": text, "voice": self._voice, "language_type": "Chinese"},
        }
        body = await self._post(payload)
        try:
            audio_url = body["output"]["audio"]["url"]
        except (KeyError, TypeError) as error:
            raise VoiceProviderError("voice upstream response missing audio url") from error
        try:
            download = await self._client.get(audio_url)
        except httpx.TimeoutException as error:
            raise VoiceProviderError(f"voice audio download timeout: {error}") from error
        except httpx.HTTPError as error:
            raise VoiceProviderError(f"voice audio download failed: {error}") from error
        if download.status_code >= 400:
            raise VoiceProviderError(
                f"voice audio download HTTP {download.status_code}"
            )
        return download.content
```

```python
"""Speech input and spoken-answer endpoints.

Both endpoints are anonymous (like ``POST /query``) because the workbench has no
browser token; abuse is controlled by ``BUCKET_VOICE`` instead. Audio is never
written to disk: it lives in request memory and is released when the request
finishes.
"""

from __future__ import annotations

import logging

from fastapi import APIRouter, Request, Response, UploadFile
from fastapi.responses import JSONResponse

from industrial_rag.errors import AppError
from industrial_rag.voice_provider import VoiceProvider, VoiceProviderError
from industrial_rag.voice_text import (
    clean_for_speech,
    is_low_confidence,
    mime_for_kind,
    sniff_audio_kind,
)

logger = logging.getLogger("industrial_rag.voice")

router = APIRouter(prefix="/v1/voice", tags=["voice"])

# A recording shorter than this is almost always an accidental tap.
_MIN_DURATION_MS = 500
_UPLOAD_CHUNK_SIZE = 256 * 1024


def _voice_settings(request: Request):
    return request.app.state.resolved_settings


def _provider(request: Request) -> VoiceProvider:
    provider = getattr(request.app.state, "voice_provider", None)
    if provider is None:
        raise AppError(
            "VOICE_DISABLED", "语音功能未启用。", status_code=404
        )
    return provider


def _ensure_enabled(request: Request) -> VoiceProvider:
    settings = _voice_settings(request)
    if not getattr(settings, "voice_enabled", False):
        raise AppError("VOICE_DISABLED", "语音功能未启用。", status_code=404)
    return _provider(request)


async def _read_limited(file: UploadFile, max_bytes: int) -> bytes:
    """Read incrementally so one request cannot exhaust process memory."""
    chunks: list[bytes] = []
    total = 0
    while chunk := await file.read(_UPLOAD_CHUNK_SIZE):
        total += len(chunk)
        if total > max_bytes:
            raise AppError(
                "VOICE_AUDIO_TOO_LARGE",
                f"录音超过 {max_bytes // (1024 * 1024)} MB 上限。",
                status_code=413,
            )
        chunks.append(chunk)
    return b"".join(chunks)


@router.post("/transcribe")
async def transcribe(
    request: Request,
    file: UploadFile,
    duration_ms: int = 0,
) -> JSONResponse:
    provider = _ensure_enabled(request)
    settings = _voice_settings(request)

    if duration_ms and duration_ms < _MIN_DURATION_MS:
        raise AppError(
            "VOICE_AUDIO_TOO_SHORT",
            "说话时间太短，请重试。",
            status_code=422,
        )
    max_seconds = int(getattr(settings, "voice_max_audio_seconds", 60))
    if duration_ms and duration_ms > max_seconds * 1000:
        raise AppError(
            "VOICE_AUDIO_TOO_LARGE",
            f"录音超过 {max_seconds} 秒上限。",
            status_code=413,
        )

    audio = await _read_limited(
        file, int(getattr(settings, "voice_max_audio_bytes", 5_242_880))
    )
    if not audio:
        raise AppError("VOICE_AUDIO_UNRECOGNISED", "未收到音频数据。", status_code=415)

    kind = sniff_audio_kind(audio)
    if kind is None:
        raise AppError(
            "VOICE_AUDIO_UNRECOGNISED",
            "无法识别音频格式，请重新录音。",
            status_code=415,
        )

    try:
        result = await provider.transcribe(audio, mime_for_kind(kind))
    except VoiceProviderError as error:
        logger.warning("voice transcribe failed: %s", error)
        raise AppError(
            "VOICE_UPSTREAM_UNAVAILABLE",
            "语音服务暂不可用，请稍后重试。",
            status_code=503,
        ) from error

    text = (result.text or "").strip()
    low = is_low_confidence(text)
    # Log the length, never the recognised text: it may contain site details.
    logger.info(
        "voice transcribe ok: %d bytes, %d chars, low_confidence=%s",
        len(audio),
        len(text),
        low,
    )
    return JSONResponse(
        {"text": text, "language": result.language, "low_confidence": low}
    )


@router.post("/speak")
async def speak(request: Request, payload: dict) -> Response:
    provider = _ensure_enabled(request)
    settings = _voice_settings(request)

    raw_text = str(payload.get("text") or "").strip()
    if not raw_text:
        raise AppError("VOICE_EMPTY_TEXT", "没有可朗读的文本。", status_code=422)

    max_chars = int(getattr(settings, "voice_tts_max_chars", 600))
    spoken = clean_for_speech(raw_text, max_chars=max_chars)
    if not spoken:
        raise AppError("VOICE_EMPTY_TEXT", "清理后没有可朗读的内容。", status_code=422)
    truncated = len(spoken) >= max_chars

    try:
        audio = await provider.synthesize(spoken)
    except VoiceProviderError as error:
        logger.warning("voice synthesize failed: %s", error)
        raise AppError(
            "VOICE_UPSTREAM_UNAVAILABLE",
            "语音合成暂不可用，请稍后重试。",
            status_code=503,
        ) from error

    headers = {"Cache-Control": "no-store"}
    if truncated:
        headers["X-Voice-Truncated"] = "true"
    return Response(content=audio, media_type="audio/wav", headers=headers)
```

```typescript
import { ref } from 'vue'

export type RecorderState =
  | 'idle'
  | 'requesting'
  | 'recording'
  | 'denied'
  | 'too-short'

export interface RecordingResult {
  blob: Blob
  durationMs: number
}

export interface UseVoiceRecorderOptions {
  minDurationMs?: number
  maxDurationMs?: number
}

/**
 * Microphone capture state machine.
 *
 * The recording is never uploaded from here: the caller decides what to do with
 * the blob. Recordings shorter than `minDurationMs` are rejected locally because
 * the ASR service hallucinates text for near-silent input instead of failing.
 */
export function useVoiceRecorder(options: UseVoiceRecorderOptions = {}) {
  const minDurationMs = options.minDurationMs ?? 500
  const maxDurationMs = options.maxDurationMs ?? 60_000

  const state = ref<RecorderState>('idle')
  const errorMessage = ref('')

  let recorder: MediaRecorder | null = null
  let stream: MediaStream | null = null
  let startedAt = 0
  let chunks: Blob[] = []

  async function start(): Promise<void> {
    state.value = 'requesting'
    errorMessage.value = ''
    try {
      stream = await navigator.mediaDevices.getUserMedia({ audio: true })
    } catch (error) {
      const name = (error as DOMException)?.name ?? ''
      state.value = 'denied'
      errorMessage.value =
        name === 'NotAllowedError'
          ? '请允许麦克风权限后再试。'
          : '无法访问麦克风，请检查设备。'
      return
    }

    chunks = []
    recorder = new MediaRecorder(stream)
    recorder.ondataavailable = (event) => {
      if (event.data.size > 0) chunks.push(event.data)
    }
    recorder.start()
    startedAt = Date.now()
    state.value = 'recording'

    if (maxDurationMs > 0) {
      setTimeout(() => {
        if (state.value === 'recording') void stop()
      }, maxDurationMs)
    }
  }

  async function stop(): Promise<RecordingResult | null> {
    if (!recorder || state.value !== 'recording') return null
    const durationMs = Date.now() - startedAt

    const finished = new Promise<Blob>((resolve) => {
      recorder!.onstop = () => resolve(new Blob(chunks, { type: 'audio/webm' }))
    })
    recorder.stop()
    const blob = await finished

    stream?.getTracks().forEach((track) => track.stop())
    stream = null
    recorder = null

    if (durationMs < minDurationMs || blob.size === 0) {
      state.value = 'too-short'
      errorMessage.value = '说话时间太短，请重试。'
      return null
    }

    state.value = 'idle'
    return { blob, durationMs }
  }

  function cancel(): void {
    if (recorder && state.value === 'recording') {
      recorder.onstop = null
      recorder.stop()
    }
    stream?.getTracks().forEach((track) => track.stop())
    stream = null
    recorder = null
    chunks = []
    state.value = 'idle'
  }

  return { state, errorMessage, start, stop, cancel }
}
```

```typescript
import { ref } from 'vue'

export type SpeechState = 'idle' | 'synthesizing' | 'playing' | 'error'

export type Synthesize = (text: string) => Promise<Blob>

/**
 * Playback state machine for spoken answers.
 *
 * Synthesis is injected rather than imported so the composable can be tested
 * without network access. Only one clip plays at a time: calling `speak` again
 * stops the previous one.
 */
export function useSpeech(synthesize: Synthesize) {
  const state = ref<SpeechState>('idle')
  const errorMessage = ref('')

  let audio: HTMLAudioElement | null = null
  let objectUrl: string | null = null

  function cleanup(): void {
    if (audio) {
      audio.onended = null
      audio.onerror = null
      audio.pause()
      audio = null
    }
    if (objectUrl) {
      URL.revokeObjectURL(objectUrl)
      objectUrl = null
    }
  }

  async function speak(text: string): Promise<void> {
    const trimmed = text.trim()
    if (!trimmed) return

    stop()
    state.value = 'synthesizing'
    errorMessage.value = ''

    let blob: Blob
    try {
      blob = await synthesize(trimmed)
    } catch {
      state.value = 'error'
      errorMessage.value = '语音合成失败，请稍后重试。'
      return
    }

    objectUrl = URL.createObjectURL(blob)
    audio = new Audio(objectUrl)
    audio.onended = () => {
      cleanup()
      state.value = 'idle'
    }
    audio.onerror = () => {
      cleanup()
      state.value = 'error'
      errorMessage.value = '音频播放失败。'
    }

    try {
      await audio.play()
      state.value = 'playing'
    } catch {
      cleanup()
      state.value = 'error'
      errorMessage.value = '浏览器阻止了音频播放。'
    }
  }

  function stop(): void {
    cleanup()
    state.value = 'idle'
  }

  return { state, errorMessage, speak, stop }
}
```
